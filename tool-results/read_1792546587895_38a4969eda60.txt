    51→export default function Home() {
    52→  const [hash, navigate] = useHashRoute()
    53→  const hydrated = useHydrated()
    54→  const wallpaper = useAppStore((s) => s.wallpaper)
    55→
    56→  const appId = useMemo(() => {
    57→    if (hash.startsWith('#/app/')) return decodeURIComponent(hash.replace('#/app/', ''))
    58→    if (hash.startsWith('#/tool/')) return decodeURIComponent(hash.replace('#/tool/', ''))
    59→    return null
    60→  }, [hash])
    61→

    63→  const showApp = hydrated && Boolean(app)
    64→  const wallpaperId = hydrated ? wallpaper : 'sunset'

    67→    (id: string) => {
    68→      useAppStore.getState().addRecent(id)
    69→      navigate(`#/app/${id}`)
    70→    },
    71→    [navigate]
    72→  )
    73→
    74→  const goHome = useCallback(() => navigate('#/'), [navigate])
    75→
    76→  const [spotlightOpen, setSpotlightOpen] = useState(false)
    77→
    78→  return (
    79→    <div
    80→      className={cn(
    81→        'wallpaper relative h-dvh w-full overflow-hidden',
    82→        WALLPAPER_CLASS[wallpaperId] ?? WALLPAPER_CLASS.sunset
    83→      )}
    84→    >
    85→      {/* subtle vignette for depth */}
    86→      <div aria-hidden className="pointer-events-none absolute inset-0 bg-black/10" />
    87→
    88→      <HomeScreen onOpenApp={openApp} onOpenSearch={() => setSpotlightOpen(true)} />
    89→
    90→      {/* Spotlight search overlay */}
    91→      <AnimatePresence>
    92→        {spotlightOpen && (
    93→          <Spotlight
    94→            key="spotlight"
    95→            onOpenApp={(id) => {
    96→              setSpotlightOpen(false)
    97→              openApp(id)
    98→            }}
    99→            onClose={() => setSpotlightOpen(false)}
   100→          />
   101→        )}
   102→      </AnimatePresence>
   103→
   104→      {/* Full-screen app window with iOS-like zoom transition */}
   105→      <AnimatePresence>
   106→        {showApp && app && (
   107→          <AppWindow key={app.id} app={app} onHome={goHome} />
   108→        )}
   109→      </AnimatePresence>
   110→    </div>
   111→  )
   112→}

   117→
   118→function HomeScreen({
   119→  onOpenApp,
   120→  onOpenSearch,
   121→}: {
   122→  onOpenApp: (id: string) => void
   123→  onOpenSearch: () => void
   124→}) {
   125→  const pagesRef = useRef<HTMLDivElement>(null)
   126→  const grid = useGridMetrics(pagesRef)
   127→  const [activePage, setActivePage] = useState(0)
   128→  const [openFolder, setOpenFolder] = useState<ToolCategoryId | null>(null)
   129→
   130→  const capacity = grid.cols * grid.rows
   131→
   132→  const pages = useMemo(() => {
   133→    const out: Tool[][] = []
   134→    for (let i = 0; i < tools.length; i += capacity) out.push(tools.slice(i, i + capacity))

   136→  }, [capacity])
   137→

   139→    () => DOCK_IDS.map((id) => getToolById(id)).filter((t): t is Tool => Boolean(t)),
   140→    []

   145→  const onScroll = useCallback((e: React.UIEvent<HTMLDivElement>) => {
   146→    const el = e.currentTarget
   147→    setActivePage(Math.max(0, Math.round(el.scrollLeft / Math.max(1, el.clientWidth))))

   151→    (index: number) => {
   152→      const el = pagesRef.current
   153→      if (!el) return
   154→      el.scrollTo({ left: index * el.clientWidth, behavior: 'smooth' })

   156→    []
   157→  )
   158→
   159→  return (
   160→    <>
   161→      <StatusBar onOpenSearch={onOpenSearch} />
   162→
   163→      {/* Pages (horizontal scroll-snap) */}
   164→      <div
   165→        ref={pagesRef}
   166→        onScroll={onScroll}
   167→        className="no-scrollbar absolute inset-0 flex snap-x snap-mandatory overflow-x-auto overflow-y-hidden pb-32 pt-[max(3.25rem,calc(env(safe-area-inset-top)+2.5rem))]"
   168→        aria-label="Home screen pages"
   169→      >
   170→        {pages.map((pageApps, i) => (
   171→          <section
   172→            key={`page-${i}`}
   173→            aria-label={`Page ${i + 1}`}
   174→            className="no-scrollbar h-full w-full shrink-0 snap-center overflow-y-auto px-5 sm:px-10"
   175→          >
   176→            <div
   177→              className="grid content-start justify-items-center gap-x-1 gap-y-4 pt-2"
   178→              style={{ gridTemplateColumns: `repeat(${grid.cols}, minmax(0, 1fr))` }}
   179→            >
   180→              {pageApps.map((t) => (
   181→                <AppIcon key={t.id} tool={t} onOpen={() => onOpenApp(t.id)} />
   182→              ))}
   183→            </div>
   184→          </section>
   185→        ))}
   186→
   187→        {/* App Library page — always the last page */}
   188→        <section
   189→          aria-label="App Library"
   190→          className="h-full w-full shrink-0 snap-center overflow-y-auto px-5 sm:px-10"
   191→        >
   192→          <AppLibrary onOpenApp={onOpenApp} onOpenFolder={setOpenFolder} />
   193→        </section>
   194→      </div>
   195→
   196→      {/* Page dots */}
   197→      <div className="absolute inset-x-0 bottom-28 z-20 flex justify-center gap-2">
   198→        {Array.from({ length: pageCount }).map((_, i) => (
   199→          <button
   200→            key={i}
   201→            onClick={() => scrollToPage(i)}
   202→            aria-label={`Go to page ${i + 1}`}
   203→            aria-current={activePage === i}
   204→            className="p-1 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-white/70 rounded-full"
   205→          >
   206→            <span
   207→              className={cn(
   208→                'block h-1.5 w-1.5 rounded-full transition-colors',
   209→                activePage === i ? 'bg-white' : 'bg-white/40 hover:bg-white/60'
   210→              )}
   211→            />
   212→          </button>
   213→        ))}
   214→      </div>
   215→
   216→      {/* Dock */}
   217→      <div className="absolute inset-x-0 bottom-0 z-20 pb-[max(1rem,env(safe-area-inset-bottom))]">
   218→        <div className="mx-auto w-fit max-w-[calc(100%-2rem)] rounded-[1.75rem] bg-white/15 px-4 py-3 shadow-lg ring-1 ring-white/25 backdrop-blur-xl">
   219→          <div className="flex gap-6">
   220→            {dockApps.map((t) => (
   221→              <AppIcon key={`dock-${t.id}`} tool={t} onOpen={() => onOpenApp(t.id)} noLabel />

   227→      {/* Folder overlay — lives OUTSIDE the scroll container so it covers the screen */}
   228→      <AnimatePresence>

   230→          <FolderOverlay
   231→            key="folder"
   232→            categoryId={openFolder}
   233→            onOpenApp={(id) => {
   234→              setOpenFolder(null)
   235→              onOpenApp(id)
   236→            }}
   237→            onClose={() => setOpenFolder(null)}
   238→          />
   239→        )}
   240→      </AnimatePresence>
   241→    </>
   242→  )
   243→}

   245→/* ================================================================== */
   246→/* Grid metrics — responsive cols/rows via ResizeObserver              */
   247→/* ================================================================== */
   248→
   249→function useGridMetrics(ref: React.RefObject<HTMLDivElement | null>) {
   250→  const [grid, setGrid] = useState({ cols: 4, rows: 4 })
   251→
   252→  useEffect(() => {
   253→    const el = ref.current
   254→    if (!el) return
   255→    const measure = () => {
   256→      const cols = Math.min(8, Math.max(4, Math.floor((el.clientWidth - 40) / 92)))
   257→      const rows = Math.min(5, Math.max(2, Math.floor((el.clientHeight - 176) / 106)))
   258→      setGrid((g) => (g.cols === cols && g.rows === rows ? g : { cols, rows }))
   259→    }
   260→    const ro = new ResizeObserver(measure)
   261→    ro.observe(el)
   262→    return () => ro.disconnect()
   263→  }, [ref])
   264→
   265→  return grid
   266→}

   268→/* ================================================================== */
   269→/* Status bar — live clock, battery, online state, search              */
   270→/* ================================================================== */
   271→
   272→function StatusBar({ onOpenSearch }: { onOpenSearch: () => void }) {
   273→  const [time, setTime] = useState<string | null>(null)
   274→  const [online, setOnline] = useState(() =>
   275→    typeof window !== 'undefined' ? navigator.onLine : true
   276→  )
   277→  const [battery, setBattery] = useState<{ level: number; charging: boolean } | null>(null)
   278→
   279→  useEffect(() => {
   280→    const update = () =>
   281→      setTime(
   282→        new Date().toLocaleTimeString('en-US', {
   283→          hour: '2-digit',
   284→          minute: '2-digit',
   285→          hour12: false,
   286→        })
   287→      )
   288→    update()
   289→    const id = setInterval(update, 15_000)
   290→    return () => clearInterval(id)
   291→  }, [])
   292→
   293→  useEffect(() => {
   294→    const goOnline = () => setOnline(true)
   295→    const goOffline = () => setOnline(false)
   296→    window.addEventListener('online', goOnline)
   297→    window.addEventListener('offline', goOffline)
   298→    return () => {
   299→      window.removeEventListener('online', goOnline)
   300→      window.removeEventListener('offline', goOffline)
   301→    }
   302→  }, [])
   303→
   304→  useEffect(() => {
   305→    interface BatteryLike extends EventTarget {
   306→      level: number
   307→      charging: boolean
   308→      addEventListener: (type: string, cb: () => void) => void
   309→      removeEventListener: (type: string, cb: () => void) => void
   310→    }
   311→    let active = true
   312→    let battery: BatteryLike | null = null
   313→    const sync = () => {
   314→      if (battery && active) setBattery({ level: battery.level, charging: battery.charging })
   315→    }
   316→    if ('getBattery' in navigator) {
   317→      ;(navigator as unknown as { getBattery: () => Promise<BatteryLike> })
   318→        .getBattery()
   319→        .then((b) => {
   320→          battery = b
   321→          sync()
   322→          b.addEventListener('levelchange', sync)
   323→          b.addEventListener('chargingchange', sync)
   324→        })
   325→        .catch(() => {})
   326→    }
   327→    return () => {
   328→      active = false
   329→      if (battery) {
   330→        battery.removeEventListener('levelchange', sync)
   331→        battery.removeEventListener('chargingchange', sync)
   332→      }
   333→    }
   334→  }, [])
   335→
   336→  return (
   337→    <header className="absolute inset-x-0 top-0 z-30 flex items-center justify-between px-6 pb-2 pt-[max(0.9rem,env(safe-area-inset-top))] text-sm font-semibold text-white drop-shadow-md">
   338→      <time suppressHydrationWarning>{time ?? '--:--'}</time>
   339→
   340→      <div className="flex items-center gap-3">
   341→        {!online && (
   342→          <span className="flex items-center gap-1" title="Offline">
   343→            <WifiOff className="h-4 w-4" aria-label="Offline" />
   344→          </span>
   345→        )}
   346→        {battery && (
   347→          <span className="flex items-center gap-1 tabular-nums">
   348→            <span className="text-xs">{Math.round(battery.level * 100)}%</span>
   349→            <BatteryGlyph level={battery.level} charging={battery.charging} />
   350→          </span>
   351→        )}
   352→        <button
   353→          onClick={onOpenSearch}
   354→          aria-label="Search apps"
   355→          className="rounded-full p-1 transition-colors hover:bg-white/15 active:scale-90"
   356→        >
   357→          <Search className="h-[18px] w-[18px]" />
   358→        </button>
   359→      </div>
   360→    </header>
   361→  )
   362→}
   363→
   364→function BatteryGlyph({ level, charging }: { level: number; charging: boolean }) {
   365→  if (charging) return <BatteryCharging className="h-4.5 w-4.5" />
   366→  if (level > 0.75) return <BatteryFull className="h-4.5 w-4.5" />
   367→  if (level > 0.35) return <BatteryMedium className="h-4.5 w-4.5" />
   368→  return <BatteryLow className="h-4.5 w-4.5" />
   369→}

   410→
   411→function Spotlight({
   412→  onOpenApp,
   413→  onClose,
   414→}: {
   415→  onOpenApp: (id: string) => void
   416→  onClose: () => void
   417→}) {
   418→  const [query, setQuery] = useState('')
   419→  const hydrated = useHydrated()
   420→  const recents = useAppStore((s) => s.recents)
   421→
   422→  const results = useMemo(() => searchTools(query, 'all'), [query])
   423→  const recentTools = useMemo(
   424→    () =>
   425→      hydrated
   426→        ? recents.map((id) => getToolById(id)).filter((t): t is Tool => Boolean(t))
   427→        : [],
   428→    [recents, hydrated]
   429→  )
   430→  const searching = query.trim().length > 0
   431→
   432→  const list = searching ? results : tools
   433→
   434→  const onKeyDown = (e: React.KeyboardEvent) => {
   435→    if (e.key === 'Escape') onClose()
   436→    if (e.key === 'Enter' && list.length > 0) onOpenApp(list[0].id)
   437→  }
   438→
   439→  return (
   440→    <motion.div
   441→      initial={{ opacity: 0, backdropFilter: 'blur(0px)' }}
   442→      animate={{ opacity: 1, backdropFilter: 'blur(24px)' }}
   443→      exit={{ opacity: 0, backdropFilter: 'blur(0px)' }}
   444→      transition={{ duration: 0.22 }}
   445→      className="absolute inset-0 z-40 flex flex-col bg-black/40 px-4 pt-[max(3rem,calc(env(safe-area-inset-top)+2rem))] backdrop-blur-2xl"
   446→      onClick={onClose}
   447→      role="dialog"
   448→      aria-label="Search apps"
   449→    >
   450→      <div className="mx-auto w-full max-w-xl" onClick={(e) => e.stopPropagation()}>
   451→        <motion.div
   452→          initial={{ y: -12, opacity: 0 }}
   453→          animate={{ y: 0, opacity: 1 }}
   454→          transition={{ delay: 0.05 }}
   455→        >
   456→          <div className="flex items-center gap-2 rounded-2xl bg-white/15 px-4 ring-1 ring-white/25 focus-within:ring-white/50">
   457→            <Search className="h-5 w-5 shrink-0 text-white/80" aria-hidden />
   458→            <input
   459→              autoFocus
   460→              value={query}
   461→              onChange={(e) => setQuery(e.target.value)}
   462→              onKeyDown={onKeyDown}
   463→              placeholder="Search apps"
   464→              aria-label="Search apps"
   465→              className="h-12 w-full bg-transparent text-base text-white placeholder:text-white/50 focus:outline-none"
   466→            />
   467→            <button
   468→              onClick={onClose}
   469→              aria-label="Close search"
   470→              className="shrink-0 rounded-full bg-white/20 px-3 py-1 text-sm font-medium text-white transition hover:bg-white/30"
   471→            >
   472→              Cancel
   473→            </button>
   474→          </div>
   475→        </motion.div>
   476→
   477→        <div className="no-scrollbar mt-4 max-h-[62vh] space-y-4 overflow-y-auto pb-8">
   478→          {/* Recents (only when not searching) */}
   479→          {!searching && recentTools.length > 0 && (
   480→            <section aria-label="Recent apps">
   481→              <h2 className="mb-2 px-1 text-xs font-semibold uppercase tracking-wide text-white/60">
   482→                Recent
   483→              </h2>
   484→              <div className="flex flex-wrap gap-2">
   485→                {recentTools.map((t) => (
   486→                  <button
   487→                    key={`recent-${t.id}`}
   488→                    onClick={() => onOpenApp(t.id)}
   489→                    className="flex items-center gap-2 rounded-full bg-white/15 py-1.5 pl-1.5 pr-4 text-sm font-medium text-white ring-1 ring-white/20 transition hover:bg-white/25 active:scale-95"
   490→                  >
   491→                    <span
   492→                      className={cn(
   493→                        'flex size-7 items-center justify-center rounded-full bg-gradient-to-br text-white',
   494→                        t.gradient
   495→                      )}
   496→                    >
   497→                      <t.icon className="h-3.5 w-3.5" aria-hidden />
   498→                    </span>
   499→                    {t.name}
   500→                  </button>
   501→                ))}
   502→              </div>
   503→            </section>
   504→          )}
   505→
   506→          {/* All / matching apps */}
   507→          <section aria-label={searching ? 'Search results' : 'All apps'}>
   508→            <h2 className="mb-2 px-1 text-xs font-semibold uppercase tracking-wide text-white/60">
   509→              {searching ? `Results for “${query.trim()}”` : 'All Apps'}
   510→            </h2>
   511→            {list.length === 0 ? (
   512→              <div className="flex flex-col items-center gap-2 py-10 text-white/70">
   513→                <SearchX className="h-8 w-8" aria-hidden />
   514→                <p className="text-sm">No apps found for “{query.trim()}”</p>
   515→              </div>
   516→            ) : (
   517→              <div className="overflow-hidden rounded-2xl bg-white/10 ring-1 ring-white/15">
   518→                {list.map((t, i) => (
   519→                  <button
   520→                    key={t.id}
   521→                    onClick={() => onOpenApp(t.id)}
   522→                    className={cn(
   523→                      'flex w-full items-center gap-3 px-3.5 py-2.5 text-left transition hover:bg-white/15 active:bg-white/20',
   524→                      i > 0 && 'border-t border-white/10'
   525→                    )}
   526→                  >
   527→                    <span
   528→                      className={cn(
   529→                        'flex size-9 shrink-0 items-center justify-center rounded-xl bg-gradient-to-br text-white shadow-sm ring-1 ring-white/20',
   530→                        t.gradient
   531→                      )}
   532→                    >
   533→                      <t.icon className="h-4.5 w-4.5" aria-hidden />
   534→                    </span>
   535→                    <span className="min-w-0 flex-1">
   536→                      <span className="block truncate text-sm font-semibold text-white">
   537→                        {t.name}
   538→                      </span>
   539→                      <span className="block truncate text-xs text-white/60">{t.description}</span>
   540→                    </span>
   541→                    <span className="shrink-0 text-[11px] font-medium uppercase tracking-wide text-white/45">
   542→                      {toolCategories.find((c) => c.id === t.category)?.name}
   543→                    </span>
   544→                  </button>
   545→                ))}
   546→              </div>
   547→            )}
   548→          </section>
   549→        </div>
   550→      </div>
   551→    </motion.div>
   552→  )
   553→}

   555→/* ================================================================== */
   556→/* App Library (last page) — auto-grouped folders                      */
   557→/* ================================================================== */
   558→
   559→function AppLibrary({
   560→  onOpenApp,
   561→  onOpenFolder,
   562→}: {
   563→  onOpenApp: (id: string) => void
   564→  onOpenFolder: (id: ToolCategoryId) => void
   565→}) {
   566→  const folders = useMemo(
   567→    () =>
   568→      toolCategories
   569→        .map((c) => ({ cat: c, items: tools.filter((t) => t.category === c.id) }))
   570→        .filter(({ items }) => items.length > 0),
   571→    []
   572→  )
   573→
   574→  return (
   575→    <div className="mx-auto max-w-2xl pt-6">
   576→      <h1 className="mb-5 text-center text-lg font-bold text-white drop-shadow-md">App Library</h1>
   577→      <div className="grid grid-cols-2 justify-items-center gap-x-3 gap-y-6 sm:grid-cols-3">
   578→        {folders.map(({ cat, items }) => (
   579→          <button
   580→            key={cat.id}
   581→            onClick={() => onOpenFolder(cat.id)}
   582→            aria-label={`Open ${cat.name} folder, ${items.length} apps`}
   583→            className="group flex w-full max-w-[150px] flex-col items-center gap-2 rounded-2xl p-1 transition-transform hover:scale-[1.03] active:scale-95 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-white/70"
   584→          >
   585→            <span className="grid w-full grid-cols-2 gap-1.5 rounded-[1.4rem] bg-white/15 p-2.5 shadow-lg ring-1 ring-white/25 backdrop-blur-xl">
   586→              {items.slice(0, 4).map((t) => (
   587→                <span
   588→                  key={t.id}
   589→                  className={cn(
   590→                    'flex aspect-square items-center justify-center rounded-[0.7rem] bg-gradient-to-br text-white shadow-sm ring-1 ring-white/20',
   591→                    t.gradient
   592→                  )}
   593→                >
   594→                  <t.icon className="h-[55%] w-[55%]" strokeWidth={1.9} aria-hidden />
   595→                </span>
   596→              ))}
   597→              {items.length > 4 && (
   598→                <span className="col-span-2 flex items-center justify-center gap-1 pt-0.5 text-[10px] font-semibold text-white/80">
   599→                  <Folder className="h-3 w-3" aria-hidden /> {items.length} apps
   600→                </span>
   601→              )}
   602→            </span>
   603→            <span className="truncate text-xs font-medium text-white drop-shadow-[0_1px_2px_rgba(0,0,0,0.5)]">
   604→              {cat.name}
   605→            </span>
   606→          </button>
   607→        ))}
   608→      </div>
   609→    </div>
   610→  )
   611→}

   613→function FolderOverlay({
   614→  categoryId,
   615→  onOpenApp,
   616→  onClose,
   617→}: {
   618→  categoryId: ToolCategoryId
   619→  onOpenApp: (id: string) => void
   620→  onClose: () => void
   621→}) {
   622→  const cat = toolCategories.find((c) => c.id === categoryId)
   623→  const items = tools.filter((t) => t.category === categoryId)
   624→
   625→  return (
   626→    <motion.div
   627→      initial={{ opacity: 0 }}
   628→      animate={{ opacity: 1 }}
   629→      exit={{ opacity: 0 }}
   630→      transition={{ duration: 0.2 }}

   637→        initial={{ scale: 0.92, opacity: 0 }}
   638→        animate={{ scale: 1, opacity: 1 }}
   639→        exit={{ scale: 0.92, opacity: 0 }}
   640→        transition={{ type: 'spring', stiffness: 380, damping: 32 }}
   641→        className="w-full max-w-md rounded-3xl bg-white/12 p-5 ring-1 ring-white/25 backdrop-blur-2xl"
   642→        onClick={(e) => e.stopPropagation()}
   643→      >
   644→        <div className="mb-4 flex items-center justify-between">
   645→          <h2 className="text-base font-bold text-white">{cat?.name}</h2>
   646→          <Button
   647→            variant="ghost"
   648→            size="icon"
   649→            onClick={onClose}
   650→            aria-label="Close folder"
   651→            className="h-8 w-8 rounded-full text-white hover:bg-white/20"

   669→
   670→function AppWindow({ app, onHome }: { app: Tool; onHome: () => void }) {
   671→  const favorites = useAppStore((s) => s.favorites)
   672→  const toggleFavorite = useAppStore((s) => s.toggleFavorite)
   673→  const isFav = favorites.includes(app.id)
   674→
   675→  useEffect(() => {
   676→    const onKey = (e: KeyboardEvent) => {
   677→      if (e.key === 'Escape') onHome()
   678→    }
   679→    window.addEventListener('keydown', onKey)
   680→    return () => window.removeEventListener('keydown', onKey)
   681→  }, [onHome])
   682→
   683→  return (
   684→    <motion.div
   685→      initial={{ opacity: 0, scale: 0.84, y: 30 }}
   686→      animate={{ opacity: 1, scale: 1, y: 0 }}
   687→      exit={{ opacity: 0, scale: 0.84, y: 30 }}
   688→      transition={{ type: 'spring', stiffness: 380, damping: 36 }}
   689→      className="absolute inset-0 z-50 flex flex-col overflow-hidden bg-background text-foreground"
   690→      role="dialog"
   691→      aria-label={app.name}
   692→    >
   693→      {/* App header */}
   694→      <header className="flex h-14 shrink-0 items-center gap-2.5 border-b bg-background/85 px-3 backdrop-blur">
   695→        <Button
   696→          variant="ghost"
   697→          size="icon"
   698→          onClick={onHome}
   699→          aria-label="Go home"
   700→          className="h-10 w-10 shrink-0 rounded-xl"
   701→        >
   702→          <ArrowLeft className="h-5 w-5" />
   703→        </Button>
   704→        <span
   705→          className={cn(
   706→            'flex size-9 shrink-0 items-center justify-center rounded-xl bg-gradient-to-br text-white shadow-sm ring-1 ring-black/5',
   707→            app.gradient
   708→          )}
   709→        >
   710→          <app.icon className="h-5 w-5" aria-hidden />
   711→        </span>
   712→        <div className="min-w-0 flex-1">
   713→          <h1 className="truncate text-base font-bold leading-tight">{app.name}</h1>
   714→          <p className="truncate text-[11px] text-muted-foreground">
   715→            {toolCategories.find((c) => c.id === app.category)?.name}
   716→          </p>
   717→        </div>
   718→        <Button
   719→          variant="ghost"
   720→          size="icon"
   721→          onClick={() => toggleFavorite(app.id)}
   722→          aria-label={isFav ? 'Remove from favorites' : 'Add to favorites'}
   723→          aria-pressed={isFav}
   724→          className="h-10 w-10 shrink-0 rounded-xl"
   725→        >
   726→          <Star className={cn('h-5 w-5', isFav && 'fill-amber-400 text-amber-500')} />
   727→        </Button>
   728→      </header>
   729→
   730→      {/* App body */}
   731→      <div className="flex-1 overflow-y-auto overscroll-contain">
   732→        <div className="mx-auto w-full max-w-4xl px-4 pb-28 pt-5">
   733→          <app.Component />
   734→        </div>
   735→      </div>
   736→
   737→      {/* Home indicator */}
   738→      <button
   739→        onClick={onHome}
   740→        aria-label="Go home"
   741→        className="group absolute bottom-[max(0.4rem,env(safe-area-inset-bottom))] left-1/2 z-10 -translate-x-1/2 p-2 focus-visible:outline-none"
   742→      >
   743→        <span className="block h-[5px] w-32 rounded-full bg-foreground/25 transition-colors group-hover:bg-foreground/50" />
   744→      </button>
   745→    </motion.div>
   746→  )
   747→}
