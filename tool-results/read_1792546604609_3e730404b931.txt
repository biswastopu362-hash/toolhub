   245→   245→/* ================================================================== */
   246→   246→/* Grid metrics — responsive cols/rows via ResizeObserver              */
   247→   247→/* ================================================================== */
   248→   248→
   249→   249→function useGridMetrics(ref: React.RefObject<HTMLDivElement | null>) {
   250→   250→  const [grid, setGrid] = useState({ cols: 4, rows: 4 })
   251→   251→
   252→   252→  useEffect(() => {
   253→   253→    const el = ref.current
   254→   254→    if (!el) return
   255→   255→    const measure = () => {
   256→   256→      const cols = Math.min(8, Math.max(4, Math.floor((el.clientWidth - 40) / 92)))
   257→   257→      const rows = Math.min(5, Math.max(2, Math.floor((el.clientHeight - 176) / 106)))
   258→   258→      setGrid((g) => (g.cols === cols && g.rows === rows ? g : { cols, rows }))
   259→   259→    }
   260→   260→    const ro = new ResizeObserver(measure)
   261→   261→    ro.observe(el)
   262→   262→    return () => ro.disconnect()
   263→   263→  }, [ref])
   264→   264→
   265→   265→  return grid
   266→   266→}

   270→   270→/* ================================================================== */
   271→   271→
   272→   272→function StatusBar({ onOpenSearch }: { onOpenSearch: () => void }) {
   273→   273→  const [time, setTime] = useState<string | null>(null)
   274→   274→  const [online, setOnline] = useState(() =>
   275→   275→    typeof window !== 'undefined' ? navigator.onLine : true
   276→   276→  )
   277→   277→  const [battery, setBattery] = useState<{ level: number; charging: boolean } | null>(null)
   278→   278→
   279→   279→  useEffect(() => {
   280→   280→    const update = () =>
   281→   281→      setTime(
   282→   282→        new Date().toLocaleTimeString('en-US', {
   283→   283→          hour: '2-digit',
   284→   284→          minute: '2-digit',
   285→   285→          hour12: false,
   286→   286→        })
   287→   287→      )
   288→   288→    update()
   289→   289→    const id = setInterval(update, 15_000)
   290→   290→    return () => clearInterval(id)
   291→   291→  }, [])
   292→   292→
   293→   293→  useEffect(() => {
   294→   294→    const goOnline = () => setOnline(true)
   295→   295→    const goOffline = () => setOnline(false)
   296→   296→    window.addEventListener('online', goOnline)
   297→   297→    window.addEventListener('offline', goOffline)
   298→   298→    return () => {
   299→   299→      window.removeEventListener('online', goOnline)
   300→   300→      window.removeEventListener('offline', goOffline)
   301→   301→    }
   302→   302→  }, [])
   303→   303→
   304→   304→  useEffect(() => {
   305→   305→    interface BatteryLike extends EventTarget {
   306→   306→      level: number
   307→   307→      charging: boolean
   308→   308→      addEventListener: (type: string, cb: () => void) => void
   309→   309→      removeEventListener: (type: string, cb: () => void) => void
   310→   310→    }
   311→   311→    let active = true
   312→   312→    let battery: BatteryLike | null = null
   313→   313→    const sync = () => {
   314→   314→      if (battery && active) setBattery({ level: battery.level, charging: battery.charging })
   315→   315→    }
   316→   316→    if ('getBattery' in navigator) {
   317→   317→      ;(navigator as unknown as { getBattery: () => Promise<BatteryLike> })
   318→   318→        .getBattery()
   319→   319→        .then((b) => {
   320→   320→          battery = b
   321→   321→          sync()
   322→   322→          b.addEventListener('levelchange', sync)
   323→   323→          b.addEventListener('chargingchange', sync)
   324→   324→        })
   325→   325→        .catch(() => {})
   326→   326→    }
   327→   327→    return () => {
   328→   328→      active = false
   329→   329→      if (battery) {
   330→   330→        battery.removeEventListener('levelchange', sync)
   331→   331→        battery.removeEventListener('chargingchange', sync)
   332→   332→      }
   333→   333→    }
   334→   334→  }, [])
   335→   335→
   336→   336→  return (
   337→   337→    <header className="absolute inset-x-0 top-0 z-30 flex items-center justify-between px-6 pb-2 pt-[max(0.9rem,env(safe-area-inset-top))] text-sm font-semibold text-white drop-shadow-md">
   338→   338→      <time suppressHydrationWarning>{time ?? '--:--'}</time>
   339→   339→
   340→   340→      <div className="flex items-center gap-3">
   341→   341→        {!online && (
   342→   342→          <span className="flex items-center gap-1" title="Offline">
   343→   343→            <WifiOff className="h-4 w-4" aria-label="Offline" />
   344→   344→          </span>
   345→   345→        )}
   346→   346→        {battery && (
   347→   347→          <span className="flex items-center gap-1 tabular-nums">
   348→   348→            <span className="text-xs">{Math.round(battery.level * 100)}%</span>
   349→   349→            <BatteryGlyph level={battery.level} charging={battery.charging} />
   350→   350→          </span>
   351→   351→        )}
   352→   352→        <button
   353→   353→          onClick={onOpenSearch}
   354→   354→          aria-label="Search apps"
   355→   355→          className="rounded-full p-1 transition-colors hover:bg-white/15 active:scale-90"
   356→   356→        >
   357→   357→          <Search className="h-[18px] w-[18px]" />
   358→   358→        </button>
   359→   359→      </div>
   360→   360→    </header>
   361→   361→  )
   362→   362→}
   363→   363→
   364→   364→function BatteryGlyph({ level, charging }: { level: number; charging: boolean }) {
   365→   365→  if (charging) return <BatteryCharging className="h-4.5 w-4.5" />
   366→   366→  if (level > 0.75) return <BatteryFull className="h-4.5 w-4.5" />
   367→   367→  if (level > 0.35) return <BatteryMedium className="h-4.5 w-4.5" />
   368→   368→  return <BatteryLow className="h-4.5 w-4.5" />
   369→   369→}

   410→   410→
   411→   411→function Spotlight({
   412→   412→  onOpenApp,
   413→   413→  onClose,
   414→   414→}: {
   415→   415→  onOpenApp: (id: string) => void
   416→   416→  onClose: () => void
   417→   417→}) {
   418→   418→  const [query, setQuery] = useState('')
   419→   419→  const hydrated = useHydrated()
   420→   420→  const recents = useAppStore((s) => s.recents)
   421→   421→
   422→   422→  const results = useMemo(() => searchTools(query, 'all'), [query])
   423→   423→  const recentTools = useMemo(
   424→   424→    () =>
   425→   425→      hydrated
   426→   426→        ? recents.map((id) => getToolById(id)).filter((t): t is Tool => Boolean(t))
   427→   427→        : [],
   428→   428→    [recents, hydrated]
   429→   429→  )
   430→   430→  const searching = query.trim().length > 0
   431→   431→
   432→   432→  const list = searching ? results : tools
   433→   433→
   434→   434→  const onKeyDown = (e: React.KeyboardEvent) => {
   435→   435→    if (e.key === 'Escape') onClose()
   436→   436→    if (e.key === 'Enter' && list.length > 0) onOpenApp(list[0].id)
   437→   437→  }
   438→   438→
   439→   439→  return (
   440→   440→    <motion.div
   441→   441→      initial={{ opacity: 0, backdropFilter: 'blur(0px)' }}
   442→   442→      animate={{ opacity: 1, backdropFilter: 'blur(24px)' }}
   443→   443→      exit={{ opacity: 0, backdropFilter: 'blur(0px)' }}
   444→   444→      transition={{ duration: 0.22 }}
   445→   445→      className="absolute inset-0 z-40 flex flex-col bg-black/40 px-4 pt-[max(3rem,calc(env(safe-area-inset-top)+2rem))] backdrop-blur-2xl"
   446→   446→      onClick={onClose}
   447→   447→      role="dialog"
   448→   448→      aria-label="Search apps"
   449→   449→    >
   450→   450→      <div className="mx-auto w-full max-w-xl" onClick={(e) => e.stopPropagation()}>
   451→   451→        <motion.div
   452→   452→          initial={{ y: -12, opacity: 0 }}
   453→   453→          animate={{ y: 0, opacity: 1 }}
   454→   454→          transition={{ delay: 0.05 }}
   455→   455→        >
   456→   456→          <div className="flex items-center gap-2 rounded-2xl bg-white/15 px-4 ring-1 ring-white/25 focus-within:ring-white/50">
   457→   457→            <Search className="h-5 w-5 shrink-0 text-white/80" aria-hidden />
   458→   458→            <input
   459→   459→              autoFocus
   460→   460→              value={query}
   461→   461→              onChange={(e) => setQuery(e.target.value)}
   462→   462→              onKeyDown={onKeyDown}
   463→   463→              placeholder="Search apps"
   464→   464→              aria-label="Search apps"
   465→   465→              className="h-12 w-full bg-transparent text-base text-white placeholder:text-white/50 focus:outline-none"
   466→   466→            />
   467→   467→            <button
   468→   468→              onClick={onClose}
   469→   469→              aria-label="Close search"
   470→   470→              className="shrink-0 rounded-full bg-white/20 px-3 py-1 text-sm font-medium text-white transition hover:bg-white/30"
   471→   471→            >
